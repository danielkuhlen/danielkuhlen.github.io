\newcommand{\sectionskip}{\par\vspace{1.5em}}  % above every section label

\newcommand{\sectionline}{\par\vspace{1em}\hrule}

% --- margin labels ----------------------------------------------------------
% Right-aligned text in the left margin, ending \colsep before the content.
% (\hspace{\leftskip} keeps labels in the margin inside the yearlist below.)
\newcommand{\marginbox}[1]{\llap{#1\hspace{\colsep}\hspace{\leftskip}}}

% Lists where wrapped lines hang (conferences, grants).
\newenvironment{hanging}{\par\everypar{\hangindent=1em\hangafter=1\relax}}{\par}

% Year column inside the content column (Conferences & Workshops):

% Events align after the year column; wrapped lines align with them.
\newlength{\yearcol}  \setlength{\yearcol}{3.2em}
\newenvironment{yearlist}
    {\par\raggedright\setlength{\leftskip}{\yearcol}}
    {\par}
\newcommand{\yearitem}[1]{%
    \ifvmode\noindent\fi\llap{\makebox[\yearcol][l]{\textit{#1}}}\ignorespaces}

% Presenter marker in the conference list (size: \footnotesize).
\newcommand{\presenter}{{\footnotesize\textbf{\textit{P}}}}
\newcommand{\discussant}{{\footnotesize\textbf{\textit{D}}}}  % discussant marker

% Section label, sharing its line with the first entry.
% Long labels can be split over lines with \\, e.g. \cvsection{Conferences \&\\ Workshops}.
\newcommand{\cvsection}[1]{%
    \sectionskip\noindent
    \marginbox{\smash{\scshape\begin{tabular}[t]{@{}r@{}}#1\end{tabular}}}\ignorespaces}

% Section label on a line of its own (for sections listed by year).
% Long labels can be split over lines with \\, e.g. \cvlabel{Conferences \&\\ Workshops}.
\newcommand{\cvlabel}[1]{%
    \sectionskip\noindent
    \marginbox{\scshape\begin{tabular}[t]{@{}r@{}}#1\end{tabular}}\strut\par}

% Italic year in the margin, sharing its line with the entry.
\newcommand{\cvyear}[1]{%
    \par\noindent\marginbox{\textit{#1}}\ignorespaces}

% --- entries ----------------------------------------------------------------
% \entry{Title}{Date}{Details}: details may be empty or span several lines (\\).
\newcommand{\entry}[3]{%
    \textbf{#1}\hfill #2\par
    \if\relax\detokenize{#3}\relax\else{\itshape #3\par}\fi}
